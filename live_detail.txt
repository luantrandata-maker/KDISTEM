-- ##### func act_norm(t text)
CREATE OR REPLACE FUNCTION public.act_norm(t text)
 RETURNS text
 LANGUAGE sql
 IMMUTABLE
AS $function$ select lower(btrim(regexp_replace(coalesce(t, ''), '\s+', ' ', 'g'))) $function$
;
-- ##### func act_shuffle(arr jsonb, seed text)
CREATE OR REPLACE FUNCTION public.act_shuffle(arr jsonb, seed text)
 RETURNS jsonb
 LANGUAGE plpgsql
 IMMUTABLE
AS $function$
declare r jsonb;
begin
  if arr is null or jsonb_typeof(arr) <> 'array' or jsonb_array_length(arr) < 2 then return coalesce(arr, '[]'::jsonb); end if;
  select jsonb_agg(v order by md5(seed || ':' || v::text || ':' || o)) into r from jsonb_array_elements(arr) with ordinality x(v, o);
  if r = arr then
    select jsonb_agg(v order by o desc) into r from jsonb_array_elements(arr) with ordinality x(v, o);
  end if;
  return r;
end $function$
;

-- ##### func activity_grade(p_kind text, c jsonb, a jsonb)
CREATE OR REPLACE FUNCTION public.activity_grade(p_kind text, c jsonb, a jsonb)
 RETURNS jsonb
 LANGUAGE plpgsql
 IMMUTABLE
AS $function$
declare res boolean[] := '{}'; e jsonb; s text; o int; by_item jsonb := '{}'::jsonb;
begin
  a := coalesce(a, 'null'::jsonb);
  case p_kind
  when 'quiz' then
    for e, o in select x, n from jsonb_array_elements(c->'items') with ordinality t(x, n) loop
      res := res || (jsonb_typeof(a->(o::int - 1)) = 'number' and (a->>(o::int - 1))::int = (e->>'answer')::int);
    end loop;
  when 'truefalse' then
    for e, o in select x, n from jsonb_array_elements(c->'items') with ordinality t(x, n) loop
      res := res || (jsonb_typeof(a->(o::int - 1)) = 'boolean' and (a->>(o::int - 1))::boolean = (e->>'answer')::boolean);
    end loop;
  when 'match' then
    for e, o in select x, n from jsonb_array_elements(c->'pairs') with ordinality t(x, n) loop
      res := res || coalesce(public.act_norm(a->>(o::int - 1)) = public.act_norm(e->>'right') and a->>(o::int - 1) is not null, false);
    end loop;
  when 'order' then
    for s, o in select x, n from jsonb_array_elements_text(c->'steps') with ordinality t(x, n) loop
      res := res || coalesce(public.act_norm(a->>(o::int - 1)) = public.act_norm(s) and a->>(o::int - 1) is not null, false);
    end loop;
  when 'fill' then
    for s, o in select m[1], n from regexp_matches(c->>'text', '\[([^\]]*)\]', 'g') with ordinality t(m, n) loop
      res := res || coalesce(a->>(o::int - 1) is not null and public.act_norm(a->>(o::int - 1)) in
        (select public.act_norm(alt) from unnest(string_to_array(s, '|')) alt), false);
    end loop;
  when 'group' then
    for e in select * from jsonb_array_elements(c->'groups') loop
      for s in select * from jsonb_array_elements_text(e->'items') loop
        res := res || coalesce(jsonb_typeof(a) = 'object' and public.act_norm(a->>s) = public.act_norm(e->>'name'), false);
        by_item := by_item || jsonb_build_object(s, res[array_length(res, 1)]);
      end loop;
    end loop;
  else
    return jsonb_build_object('correct', 0, 'total', 0, 'results', '[]'::jsonb);
  end case;
  return jsonb_build_object('correct', (select count(*) from unnest(res) r where r), 'total', coalesce(array_length(res, 1), 0), 'results', to_jsonb(res))
    || case when p_kind = 'group' then jsonb_build_object('by_item', by_item) else '{}'::jsonb end;
end $function$
;
-- ##### func activity_playable(p_kind text, c jsonb, seed text)
CREATE OR REPLACE FUNCTION public.activity_playable(p_kind text, c jsonb, seed text)
 RETURNS jsonb
 LANGUAGE plpgsql
 IMMUTABLE
AS $function$
begin
  return case p_kind
    when 'quiz' then jsonb_build_object('items', (select coalesce(jsonb_agg(jsonb_build_object('q', e->'q', 'options', e->'options') order by o), '[]') from jsonb_array_elements(c->'items') with ordinality x(e, o)))
    when 'truefalse' then jsonb_build_object('items', (select coalesce(jsonb_agg(jsonb_build_object('q', e->'q') order by o), '[]') from jsonb_array_elements(c->'items') with ordinality x(e, o)))
    when 'match' then jsonb_build_object(
      'left', (select coalesce(jsonb_agg(e->'left' order by o), '[]') from jsonb_array_elements(c->'pairs') with ordinality x(e, o)),
      'right', public.act_shuffle((select coalesce(jsonb_agg(e->'right' order by o), '[]') from jsonb_array_elements(c->'pairs') with ordinality x(e, o)), seed))
    when 'order' then jsonb_build_object('steps', public.act_shuffle(c->'steps', seed))
    when 'fill' then jsonb_build_object('pieces', to_jsonb(regexp_split_to_array(c->>'text', '\[[^\]]*\]')))
    when 'group' then jsonb_build_object(
      'groups', (select coalesce(jsonb_agg(g->'name' order by o), '[]') from jsonb_array_elements(c->'groups') with ordinality x(g, o)),
      'items', public.act_shuffle((select coalesce(jsonb_agg(to_jsonb(s)), '[]') from jsonb_array_elements(c->'groups') g, jsonb_array_elements_text(g->'items') s), seed))
    when 'flashcard' then jsonb_build_object('cards', c->'cards')
    when 'worksheet' then jsonb_build_object('intro', c->'intro', 'fields', c->'fields')
    when 'lesson' then jsonb_build_object('slides', (select coalesce(jsonb_agg(e - 'answer' order by o), '[]') from jsonb_array_elements(c->'slides') with ordinality x(e, o)))
    else c end;
end $function$
;
-- ##### func activity_problem(p_kind text, c jsonb)
CREATE OR REPLACE FUNCTION public.activity_problem(p_kind text, c jsonb)
 RETURNS text
 LANGUAGE plpgsql
 IMMUTABLE
AS $function$
declare e jsonb; i int := 0; n int;
begin
  if c is null or jsonb_typeof(c) <> 'object' then return 'Nội dung không hợp lệ'; end if;
  case p_kind
  when 'quiz' then
    if jsonb_typeof(c->'items') <> 'array' or jsonb_array_length(c->'items') not between 1 and 50 then return 'Cần 1–50 câu hỏi'; end if;
    for e in select * from jsonb_array_elements(c->'items') loop
      i := i + 1;
      if public.act_norm(e->>'q') = '' then return format('Câu %s chưa có nội dung', i); end if;
      if jsonb_typeof(e->'options') <> 'array' or jsonb_array_length(e->'options') not between 2 and 6 then return format('Câu %s cần 2–6 phương án', i); end if;
      if exists (select 1 from jsonb_array_elements_text(e->'options') o where public.act_norm(o) = '') then return format('Câu %s có phương án trống', i); end if;
      if jsonb_typeof(e->'answer') <> 'number' or (e->>'answer')::int not between 0 and jsonb_array_length(e->'options') - 1 then return format('Câu %s chưa chọn đáp án', i); end if;
    end loop;
  when 'truefalse' then
    if jsonb_typeof(c->'items') <> 'array' or jsonb_array_length(c->'items') not between 1 and 50 then return 'Cần 1–50 nhận định'; end if;
    for e in select * from jsonb_array_elements(c->'items') loop
      i := i + 1;
      if public.act_norm(e->>'q') = '' then return format('Nhận định %s chưa có nội dung', i); end if;
      if jsonb_typeof(e->'answer') <> 'boolean' then return format('Nhận định %s chưa chọn Đúng/Sai', i); end if;
    end loop;
  when 'match' then
    if jsonb_typeof(c->'pairs') <> 'array' or jsonb_array_length(c->'pairs') not between 2 and 20 then return 'Cần 2–20 cặp'; end if;
    for e in select * from jsonb_array_elements(c->'pairs') loop
      i := i + 1;
      if public.act_norm(e->>'left') = '' or public.act_norm(e->>'right') = '' then return format('Cặp %s còn trống', i); end if;
    end loop;
    if (select count(distinct public.act_norm(x->>'right')) from jsonb_array_elements(c->'pairs') x) < jsonb_array_length(c->'pairs') then
      return 'Vế phải của các cặp phải khác nhau'; end if;
  when 'order' then
    if jsonb_typeof(c->'steps') <> 'array' or jsonb_array_length(c->'steps') not between 2 and 20 then return 'Cần 2–20 bước'; end if;
    if exists (select 1 from jsonb_array_elements_text(c->'steps') s where public.act_norm(s) = '') then return 'Có bước còn trống'; end if;
    if (select count(distinct public.act_norm(s)) from jsonb_array_elements_text(c->'steps') s) < jsonb_array_length(c->'steps') then
      return 'Các bước phải khác nhau'; end if;
  when 'fill' then
    if public.act_norm(c->>'text') = '' then return 'Chưa có đoạn văn'; end if;
    select count(*) into n from regexp_matches(c->>'text', '\[([^\]]*)\]', 'g') m;
    if n = 0 then return 'Đánh dấu chỗ trống bằng [đáp án]'; end if;
    if n > 30 then return 'Tối đa 30 chỗ trống'; end if;
    if exists (select 1 from regexp_matches(c->>'text', '\[([^\]]*)\]', 'g') m where public.act_norm(m[1]) = '') then return 'Có chỗ trống chưa ghi đáp án'; end if;
  when 'group' then
    if jsonb_typeof(c->'groups') <> 'array' or jsonb_array_length(c->'groups') not between 2 and 6 then return 'Cần 2–6 nhóm'; end if;
    for e in select * from jsonb_array_elements(c->'groups') loop
      i := i + 1;
      if public.act_norm(e->>'name') = '' then return format('Nhóm %s chưa có tên', i); end if;
      if jsonb_typeof(e->'items') <> 'array' or jsonb_array_length(e->'items') = 0 then return format('Nhóm %s chưa có thẻ', i); end if;
      if exists (select 1 from jsonb_array_elements_text(e->'items') s where public.act_norm(s) = '') then return format('Nhóm %s có thẻ trống', i); end if;
    end loop;
    select count(*) into n from jsonb_array_elements(c->'groups') g, jsonb_array_elements_text(g->'items') s;
    if n > 40 then return 'Tối đa 40 thẻ'; end if;
    if (select count(distinct public.act_norm(s)) from jsonb_array_elements(c->'groups') g, jsonb_array_elements_text(g->'items') s) < n then
      return 'Các thẻ phải khác nhau'; end if;
  when 'flashcard' then
    if jsonb_typeof(c->'cards') <> 'array' or jsonb_array_length(c->'cards') not between 1 and 60 then return 'Cần 1–60 thẻ'; end if;
    if exists (select 1 from jsonb_array_elements(c->'cards') x where public.act_norm(x->>'front') = '' or public.act_norm(x->>'back') = '') then return 'Có thẻ còn trống mặt trước hoặc mặt sau'; end if;
  when 'worksheet' then
    if jsonb_typeof(c->'fields') <> 'array' or jsonb_array_length(c->'fields') not between 1 and 30 then return 'Cần 1–30 mục trả lời'; end if;
    if exists (select 1 from jsonb_array_elements(c->'fields') x where public.act_norm(x->>'label') = '' or coalesce(x->>'type', 'text') not in ('text','long','number')) then
      return 'Có mục trả lời chưa đặt tên'; end if;
  else return 'Loại hoạt động không hợp lệ';
  end case;
  return null;
end $function$
;
-- ##### func apply_invitations(p_user uuid)
CREATE OR REPLACE FUNCTION public.apply_invitations(p_user uuid)
 RETURNS integer
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare v_email text; v_confirmed timestamptz; inv invitations; n int := 0;
begin
  select email, email_confirmed_at into v_email, v_confirmed from auth.users where id = p_user;
  if v_email is null or v_confirmed is null then return 0; end if;
  for inv in select * from invitations where lower(email) = lower(v_email) and status = 'pending' order by created_at loop
    insert into memberships (user_id, org_id, role_id, created_by)
      values (p_user, inv.org_id, inv.role_id, inv.invited_by) on conflict do nothing;
    update profiles set full_name = coalesce(nullif(trim(inv.full_name), ''), full_name)
     where id = p_user and (org_id = inv.org_id or full_name is null or full_name = split_part(v_email, '@', 1));
    if inv.teacher_id is not null then
      update teachers set profile_id = p_user where id = inv.teacher_id and (profile_id is null or profile_id = p_user);
    end if;
    if inv.student_id is not null then
      update students set profile_id = p_user where id = inv.student_id and (profile_id is null or profile_id = p_user);
    end if;
    update invitations set status = 'accepted', accepted_at = now(), accepted_user = p_user, note = null where id = inv.id;
    n := n + 1;
  end loop;
  return n;
end $function$
;
-- ##### func approval_apply(p_request uuid, p_decision text, p_note text, p_sync boolean)
CREATE OR REPLACE FUNCTION public.approval_apply(p_request uuid, p_decision text, p_note text, p_sync boolean)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare r approval_requests; n int; v_final text; st jsonb;
begin
  select * into r from approval_requests where id = p_request for update;
  if r.id is null then raise exception 'Không tìm thấy hồ sơ'; end if;
  if r.status <> 'pending' then raise exception 'Hồ sơ đã được xử lý' using errcode = '55000'; end if;
  if p_decision not in ('approve', 'reject', 'comment') then raise exception 'Quyết định không hợp lệ'; end if;
  n := jsonb_array_length(r.steps);
  st := r.steps -> (r.current_step - 1);
  -- Người duyệt bước hiện tại; riêng TỪ CHỐI thì Ban giám hiệu / quản trị của đơn vị gửi luôn làm được
  if not public.approval_can_act(r.id)
     and not (p_decision = 'reject' and public.my_org_id() = r.org_id and public.is_admin()) then
    raise exception 'Bạn không phải người duyệt bước "%"', st->>'name' using errcode = '42501';
  end if;
  insert into approval_actions (request_id, step_no, actor, decision, note) values (r.id, r.current_step, auth.uid(), p_decision, nullif(trim(p_note), ''));
  if p_decision = 'comment' then return jsonb_build_object('status', 'pending', 'step', r.current_step, 'steps', n); end if;

  if p_decision = 'approve' and r.current_step < n then
    update approval_requests set current_step = current_step + 1 where id = r.id;
    perform public.approval_notify_step(r.id);
    return jsonb_build_object('status', 'pending', 'step', r.current_step + 1, 'steps', n,
                              'next', r.steps -> r.current_step ->> 'name', 'next_org', r.steps -> r.current_step ->> 'org_name');
  end if;

  v_final := case p_decision when 'approve' then 'approved' else 'rejected' end;
  update approval_requests set status = v_final, decided_at = now() where id = r.id;
  if p_sync then
    perform set_config('app.approval_engine', 'on', true);
    execute format('update %I set status = $1 where id = $2', r.ref_table) using v_final, r.ref_id;
    perform set_config('app.approval_engine', 'off', true);
  end if;
  -- Báo người gửi (đăng ký STEM, phiếu mượn đã có thông báo riêng khi đổi trạng thái)
  if r.kind not in ('stem_registration', 'borrow_request') then
    perform public.notify_user(r.requested_by, r.org_id, 'status',
      public.approval_kind_label(r.kind) || case v_final when 'approved' then ' đã được duyệt' else ' bị từ chối' end,
      r.title || coalesce(' — ' || nullif(trim(p_note), ''), ''), 'Hộp phê duyệt', null, r.id);
  end if;
  return jsonb_build_object('status', v_final, 'step', r.current_step, 'steps', n);
end $function$
;
-- ##### func approval_build_steps(p_org uuid, p_kind text, p_workflow uuid)
CREATE OR REPLACE FUNCTION public.approval_build_steps(p_org uuid, p_kind text, p_workflow uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare s record; v jsonb := '[]'::jsonb; v_org uuid; n int := 0;
begin
  if p_workflow is not null then
    for s in select st.*, r.name as role_name from approval_steps st left join roles r on r.id = st.role_id
              where st.workflow_id = p_workflow order by st.step_no loop
      v_org := case when s.org_type is null then p_org else public.org_ancestor_of_type(p_org, s.org_type) end;
      continue when v_org is null;
      n := n + 1;
      v := v || jsonb_build_object('no', n, 'name', s.name, 'approver', s.approver, 'role_id', s.role_id, 'role_name', s.role_name,
                                   'org_id', v_org, 'org_name', (select name from organizations where id = v_org));
    end loop;
  end if;
  if n = 0 then
    v := jsonb_build_array(jsonb_build_object('no', 1, 'name', 'Duyệt', 'approver', 'perm', 'role_id', null, 'role_name', null,
                                              'org_id', p_org, 'org_name', (select name from organizations where id = p_org)));
  end if;
  return v;
end $function$
;
-- ##### func approval_can_act(p_request uuid)
CREATE OR REPLACE FUNCTION public.approval_can_act(p_request uuid)
 RETURNS boolean
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare r approval_requests; st jsonb; v_org uuid;
begin
  select * into r from approval_requests where id = p_request;
  if r.id is null or r.status <> 'pending' then return false; end if;
  st := r.steps -> (r.current_step - 1);
  v_org := (st->>'org_id')::uuid;
  if v_org is distinct from public.my_org_id() then return false; end if;
  if st->>'approver' = 'role' then
    return exists (select 1 from memberships m where m.user_id = auth.uid() and m.org_id = v_org and m.role_id = (st->>'role_id')::uuid);
  end if;
  return public.has_perm(public.approval_kind_module(r.kind), 'edit');
end $function$
;
-- ##### func approval_cancel(p_request uuid, p_note text)
CREATE OR REPLACE FUNCTION public.approval_cancel(p_request uuid, p_note text DEFAULT NULL::text)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare r approval_requests;
begin
  select * into r from approval_requests where id = p_request for update;
  if r.requested_by is distinct from auth.uid() or r.status <> 'pending' then raise exception 'Không rút được hồ sơ này' using errcode = '42501'; end if;
  insert into approval_actions (request_id, step_no, actor, decision, note) values (r.id, r.current_step, auth.uid(), 'cancel', p_note);
  update approval_requests set status = 'cancelled', decided_at = now() where id = r.id;
  perform set_config('app.approval_engine', 'on', true);
  -- Đề xuất, kế hoạch: về nháp để sửa và gửi lại; đăng ký dạy, phiếu mượn: coi như huỷ (không còn kẹt ở "chờ duyệt")
  execute format('update %I set status = $1 where id = $2', r.ref_table)
    using case when r.ref_table in ('package_orders', 'education_plans') then 'draft' else 'rejected' end, r.ref_id;
  perform set_config('app.approval_engine', 'off', true);
end $function$
;

-- ##### func approval_decide_ref(p_table text, p_ref uuid, p_decision text, p_note text)
CREATE OR REPLACE FUNCTION public.approval_decide_ref(p_table text, p_ref uuid, p_decision text, p_note text DEFAULT NULL::text)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$

begin
  select id into v from approval_requests where ref_table = p_table and ref_id = p_ref and status = 'pending';
  if v is null then raise exception 'Không có hồ sơ chờ duyệt' using errcode = '55000'; end if;
  return public.approval_apply(v, p_decision, p_note, true);
end $function$
;
-- ##### func approval_effective_workflow(p_org uuid, p_kind text)
CREATE OR REPLACE FUNCTION public.approval_effective_workflow(p_org uuid, p_kind text)
 RETURNS uuid
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  with recursive up as (
    select id, parent_id, 0 as d from organizations where id = p_org
    union all
    select o.id, o.parent_id, up.d + 1 from organizations o join up on o.id = up.parent_id where up.d < 10
  )
  select w.id from up join approval_workflows w on w.org_id = up.id and w.kind = p_kind and w.is_active
   where up.d = 0 or w.applies_to_children
   order by up.d limit 1
$function$
;
-- ##### func approval_inbox()
CREATE OR REPLACE FUNCTION public.approval_inbox()
 RETURNS SETOF approval_requests
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  select r.* from approval_requests r
   where r.status = 'pending'
     and (r.steps -> (r.current_step - 1) ->> 'org_id')::uuid = public.my_org_id()
     and public.approval_can_act(r.id)
   order by r.created_at
$function$
;

-- ##### func approval_notify_step(p_request uuid)
CREATE OR REPLACE FUNCTION public.approval_notify_step(p_request uuid)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare r approval_requests; st jsonb; v_org uuid; v_portal text; u uuid;
begin
  select * into r from approval_requests where id = p_request;
  st := r.steps -> (r.current_step - 1);
  v_org := (st->>'org_id')::uuid;
  v_portal := case when (select type from organizations where id = v_org) = 'school'
                   then coalesce((select portal from roles where id = (st->>'role_id')::uuid), 'school_admin') else 'department' end;
  for u in select distinct m.user_id from memberships m
            where m.org_id = v_org
              and case when st->>'approver' = 'role' then m.role_id = (st->>'role_id')::uuid
                       else public.user_has_perm(m.user_id, public.approval_kind_module(r.kind), 'edit', v_org) end loop
    perform public.notify_user(u, v_org, 'approval', public.approval_kind_label(r.kind) || ' chờ duyệt'
      || case when jsonb_array_length(r.steps) > 1 then format(' (bước %s/%s)', r.current_step, jsonb_array_length(r.steps)) else '' end,
      r.title, 'Hộp phê duyệt', v_portal, r.id);
  end loop;
end $function$

-- ##### func approval_save_workflow(p_kind text, p_name text, p_applies boolean, p_steps jsonb)
CREATE OR REPLACE FUNCTION public.approval_save_workflow(p_kind text, p_name text, p_applies boolean, p_steps jsonb)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare v_org uuid := public.my_org_id(); v_wf uuid; s jsonb; i int := 0; v_type text;
begin
  if not (public.is_admin() or public.is_authority()) then raise exception 'Chỉ quản trị đơn vị được đặt quy trình' using errcode = '42501'; end if;
  if jsonb_typeof(p_steps) <> 'array' or jsonb_array_length(p_steps) = 0 then raise exception 'Quy trình cần ít nhất một bước'; end if;
  select type into v_type from organizations where id = v_org;
  insert into approval_workflows (org_id, kind, name, applies_to_children, updated_at, updated_by)
    values (v_org, p_kind, coalesce(nullif(trim(p_name), ''), public.approval_kind_label(p_kind)), coalesce(p_applies, false) and v_type <> 'school', now(), auth.uid())
    on conflict (org_id, kind) do update set name = excluded.name, applies_to_children = excluded.applies_to_children,
      is_active = true, updated_at = now(), updated_by = auth.uid()
    returning id into v_wf;
  delete from approval_steps where workflow_id = v_wf;
  for s in select * from jsonb_array_elements(p_steps) loop
    i := i + 1;
    if s->>'approver' = 'role' and not exists (
         select 1 from roles r where r.id = (s->>'role_id')::uuid and (r.org_id is null or r.org_id = v_org)) then
      raise exception 'Vai trò ở bước % không hợp lệ', i;
    end if;
    insert into approval_steps (workflow_id, step_no, name, approver, role_id, org_type)
      values (v_wf, i, coalesce(nullif(trim(s->>'name'), ''), 'Bước ' || i), s->>'approver',
              case when s->>'approver' = 'role' then (s->>'role_id')::uuid end, nullif(s->>'org_type', ''));
  end loop;
  return v_wf;
end $function$
;

-- ##### func attendance_summary(p_from date, p_to date)
CREATE OR REPLACE FUNCTION public.attendance_summary(p_from date, p_to date)
 RETURNS TABLE(org_id uuid, school text, ward text, parent_name text, planned integer, taught integer, not_marked integer, cancelled integer, makeup integer, stem_planned integer, stem_taught integer, stem_room_periods integer, present integer, absent integer, late integer, excused integer, teachers_not_marked integer)
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
begin
  if not public.is_staff() then raise exception 'Không có quyền' using errcode = '42501'; end if;
  if p_from is null or p_to is null or p_to < p_from or p_to - p_from > 400 then
    raise exception 'Khoảng ngày không hợp lệ' using errcode = '22023';
  end if;
  return query
  with sc as (
    select o.id, o.name, o.ward, par.name as parent_name from organizations o
    left join organizations par on par.id = o.parent_id
    where o.type = 'school' and o.id in (select public.my_org_tree())
  ), s as (
    select ts.*, (ts.session_date + p.starts_at) at time zone 'Asia/Ho_Chi_Minh' as starts_at
    from teaching_sessions ts join period_configs p on p.id = ts.period_id
    where ts.org_id in (select id from sc) and ts.session_date between p_from and p_to
  )
  select sc.id, sc.name, sc.ward, sc.parent_name,
    count(s.id) filter (where s.status <> 'cancelled' and s.starts_at <= now())::int,
    count(s.id) filter (where s.status = 'taught')::int,
    count(s.id) filter (where s.status = 'scheduled' and s.starts_at <= now())::int,
    count(s.id) filter (where s.status = 'cancelled')::int,
    count(s.id) filter (where s.is_makeup and s.status <> 'cancelled')::int,
    count(s.id) filter (where s.kind = 'stem' and s.status <> 'cancelled' and s.starts_at <= now())::int,
    count(s.id) filter (where s.kind = 'stem' and s.status = 'taught')::int,
    count(s.id) filter (where s.room_id is not null and s.status = 'taught')::int,
    coalesce(sum(s.present_count), 0)::int, coalesce(sum(s.absent_count), 0)::int,
    coalesce(sum(s.late_count), 0)::int, coalesce(sum(s.excused_count), 0)::int,
    count(distinct s.teacher_id) filter (where s.status = 'scheduled' and s.starts_at <= now())::int
  from sc left join s on s.org_id = sc.id
  group by sc.id, sc.name, sc.ward, sc.parent_name
  order by sc.name;
end $function$
;

-- ##### func booking_put(p_org uuid, p_room uuid, p_holder uuid, p_during tstzrange, p_source text, p_source_id uuid, p_label text, p_soft boolean)
CREATE OR REPLACE FUNCTION public.booking_put(p_org uuid, p_room uuid, p_holder uuid, p_during tstzrange, p_source text, p_source_id uuid, p_label text, p_soft boolean)
 RETURNS boolean
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare b resource_bookings; v_room text;
begin
  begin
    insert into resource_bookings (org_id, room_id, holder_id, during, source, source_id, label)
    values (p_org, p_room, p_holder, p_during, p_source, p_source_id, p_label);
    delete from booking_conflicts where source = p_source and source_id = p_source_id;
    return true;
  exception when exclusion_violation then
    select * into b from resource_bookings
     where room_id = p_room and during && p_during and holder_id <> p_holder
     order by lower(during) limit 1;
    if p_soft then
      insert into booking_conflicts (org_id, room_id, holder_id, during, source, source_id, label, blocking_source, blocking_id, blocking_label)
      values (p_org, p_room, p_holder, p_during, p_source, p_source_id, p_label, b.source, b.source_id, b.label)
      on conflict (source, source_id) do update set room_id = excluded.room_id, holder_id = excluded.holder_id,
        during = excluded.during, label = excluded.label, blocking_source = excluded.blocking_source,
        blocking_id = excluded.blocking_id, blocking_label = excluded.blocking_label, detected_at = now();
      return false;
    end if;
    select name into v_room from stem_rooms where id = p_room;
    raise exception '% đã có lịch lúc %: %', coalesce(v_room, 'Phòng'), public.slot_text(b.during), coalesce(b.label, 'tiết khác')
      using errcode = '23P01', hint = 'Chọn phòng hoặc tiết khác, hoặc huỷ lịch đang giữ phòng.';
  end;
end $function$
;
-- ##### func booking_refresh_borrow(p_id uuid)
CREATE OR REPLACE FUNCTION public.booking_refresh_borrow(p_id uuid)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare b borrow_requests; v_label text;
begin
  delete from resource_bookings where source = 'borrow_request' and source_id = p_id;
  delete from booking_conflicts where source = 'borrow_request' and source_id = p_id;
  select * into b from borrow_requests where id = p_id;
  if found and b.mode = 'in_room' and b.room_id is not null and b.status in ('pending', 'approved', 'lent') then
    select concat_ws(' · ', 'Phiếu mượn', c.name, b.title) into v_label from (select 1) x left join classes c on c.id = b.class_id;
    perform public.booking_put(b.org_id, b.room_id, coalesce(b.class_id, b.teacher_id, b.id), public.period_slot(b.on_date, b.period_id),
      'borrow_request', b.id, v_label, coalesce(current_setting('kdi.booking_soft', true), '') = 'on');
  end if;
  perform public.booking_retry('borrow_request', p_id);
end $function$
;

-- ##### func booking_refresh_session(p_id uuid)

-- ##### func booking_sync()
CREATE OR REPLACE FUNCTION public.booking_sync()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare v_id uuid := coalesce(new.id, old.id);
begin
  if tg_op = 'UPDATE' then
    -- Chỉ đồng bộ lại khi đổi thứ liên quan đến lịch (điểm danh, ghi chú… không đụng sổ giữ chỗ)
    -- (IF lồng nhau: mỗi bảng có cột khác nhau, PL/pgSQL không được đọc cột của bảng khác)
    if tg_table_name = 'teaching_sessions' then
      if (old.room_id, old.class_id, old.session_date, old.period_id, old.status = 'cancelled', old.is_makeup, old.timetable_entry_id)
         is not distinct from (new.room_id, new.class_id, new.session_date, new.period_id, new.status = 'cancelled', new.is_makeup, new.timetable_entry_id) then return null; end if;
    elsif tg_table_name = 'stem_registrations' then
      if (old.room_id, old.class_id, old.on_date, old.period_id, old.status in ('pending', 'approved'))
         is not distinct from (new.room_id, new.class_id, new.on_date, new.period_id, new.status in ('pending', 'approved')) then return null; end if;
    else
      if (old.room_id, old.class_id, old.on_date, old.period_id, old.mode, old.status in ('pending', 'approved', 'lent'))
         is not distinct from (new.room_id, new.class_id, new.on_date, new.period_id, new.mode, new.status in ('pending', 'approved', 'lent')) then return null; end if;
    end if;
  end if;
  if tg_table_name = 'teaching_sessions' then perform public.booking_refresh_session(v_id);
  elsif tg_table_name = 'stem_registrations' then perform public.booking_refresh_registration(v_id);
  else perform public.booking_refresh_borrow(v_id); end if;
  return null;
end $function$
;
-- ##### func borrow_stock_guard()
CREATE OR REPLACE FUNCTION public.borrow_stock_guard()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare v_msg text;
begin
  if new.status not in ('approved', 'lent') then return new; end if;
  if tg_op = 'UPDATE' and old.status in ('approved', 'lent')
     and (old.on_date, old.period_id, old.item_list) is not distinct from (new.on_date, new.period_id, new.item_list) then return new; end if;
  -- Xếp hàng theo trường: hai người duyệt cùng lúc không cùng lấy một thiết bị
  perform pg_advisory_xact_lock(hashtext('borrow_stock'), hashtext(new.org_id::text));
  select string_agg(format('%s: cần %s, dùng được %s, đã cho mượn %s', s.name, s.need, s.usable, s.held), '; ') into v_msg
    from public.equipment_shortage(new.org_id, new.on_date, new.period_id, new.item_list, new.id) s
   where s.need > s.usable - s.held;
  if v_msg is not null then
    raise exception 'Không đủ thiết bị cho khung giờ này — %', v_msg
      using errcode = '23P01', hint = 'Đổi ngày/tiết, giảm số lượng hoặc chờ phiếu khác trả thiết bị.';
  end if;
  return new;
end $function$
;

-- ##### func dept_school_stats()
CREATE OR REPLACE FUNCTION public.dept_school_stats()
 RETURNS TABLE(school_id uuid, code text, name text, level text, ward text, activated boolean, teachers integer, stem_teachers integer, students integer, classes integer, stem_classes integer, stem_students integer, programs text[], rooms integer, rooms_active integer, stem_periods integer, total_periods integer, room_periods integer, lesson_plans integer, products integer, trained integer, training_done integer, attendance_rate numeric, units integer, units_broken integer, units_maintenance integer, units_available integer, graded integer, avg_score numeric, k12_synced boolean, has_year boolean, last_activity timestamp with time zone, kpi integer)
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  with s as (
    select o.* from organizations o
    where o.type = 'school' and public.is_staff() and public.can_access_org(o.id)
  ),
  cur as (select org_id, id from academic_years where is_current),
  base as (
    select s.id as school_id, s.code, s.name, s.level, s.ward, s.activated_at is not null as activated,
      (select count(*) from teachers t where t.org_id = s.id and t.status = 'active')::int as teachers,
      (select count(distinct teacher_id) from stem_teacher_programs x where x.org_id = s.id)::int as stem_teachers,
      (select count(*) from students x where x.org_id = s.id and coalesce(x.status,'studying') = 'studying')::int as students,
      (select count(*) from classes c where c.org_id = s.id)::int as classes,
      (select count(distinct class_id) from class_programs cp where cp.org_id = s.id)::int as stem_classes,
      (select count(*) from students x where x.org_id = s.id and coalesce(x.status,'studying') = 'studying' and x.class_id in (select class_id from class_programs cp where cp.org_id = s.id))::int as stem_students,
      coalesce((select array_agg(distinct program order by program) from class_programs cp where cp.org_id = s.id), '{}') as programs,
      (select count(*) from stem_rooms r where r.org_id = s.id)::int as rooms,
      (select count(*) from stem_rooms r where r.org_id = s.id and r.status = 'active')::int as rooms_active,
      (select count(*) from timetable_entries e where e.org_id = s.id and e.kind = 'stem'
         and (e.year_id is null or e.year_id in (select id from cur where cur.org_id = s.id)))::int as stem_periods,
      (select count(*) from timetable_entries e where e.org_id = s.id
         and (e.year_id is null or e.year_id in (select id from cur where cur.org_id = s.id)))::int as total_periods,
      (select count(*) from timetable_entries e where e.org_id = s.id and e.room_id is not null
         and (e.year_id is null or e.year_id in (select id from cur where cur.org_id = s.id)))::int as room_periods,
      (select count(*) from lesson_plans lp where lp.org_id = s.id and not lp.is_template and lp.status = 'published')::int as lesson_plans,
      (select count(*) from assignments a where a.org_id = s.id and a.kind = 'product')::int as products,
      (select count(distinct en.teacher_id) from training_enrollments en join teachers t on t.id = en.teacher_id where t.org_id = s.id)::int as trained,
      (select count(distinct en.teacher_id) from training_enrollments en join teachers t on t.id = en.teacher_id where t.org_id = s.id and en.completed)::int as training_done,
      (select round(100.0 * count(*) filter (where ta.present) / nullif(count(*) filter (where ta.present is not null), 0), 1)
         from training_attendance ta join teachers t on t.id = ta.teacher_id where t.org_id = s.id) as attendance_rate,
      (select count(*) from equipment_units u where u.org_id = s.id)::int as units,
      (select count(*) from equipment_units u where u.org_id = s.id and u.status = 'broken')::int as units_broken,
      (select count(*) from equipment_units u where u.org_id = s.id and u.status = 'maintenance')::int as units_maintenance,
      (select count(*) from equipment_units u where u.org_id = s.id and u.status = 'available')::int as units_available,
      (select count(*) from submissions x where x.org_id = s.id and x.status = 'graded')::int as graded,
      (select round(avg(x.score / nullif(a.max_score, 0) * 10), 2) from submissions x join assignments a on a.id = x.assignment_id
         where x.org_id = s.id and x.status = 'graded') as avg_score,
      exists (select 1 from teachers t where t.org_id = s.id and t.k12_synced_at is not null) as k12_synced,
      exists (select 1 from academic_years y where y.org_id = s.id) as has_year,
      greatest((select max(created_at) from stem_registrations x where x.org_id = s.id),
               (select max(submitted_at) from submissions x where x.org_id = s.id),
               (select max(started_at) from class_sessions x where x.org_id = s.id),
               (select max(updated_at) from lesson_plans x where x.org_id = s.id)) as last_activity
    from s
  )
  -- KPI tổng hợp 0–100: độ phủ CT (25) + phòng STEM (15) + tiết STEM/lớp (20) + tập huấn (20) + giáo án (10) + đồng bộ K12 (10)
  select b.*,
    round(
      25 * least(coalesce(array_length(b.programs, 1), 0), 5) / 5.0
    + 15 * least(b.rooms_active, 2) / 2.0
    + 20 * least(b.stem_periods::numeric / nullif(b.classes, 0), 2) / 2.0
    + 20 * coalesce(b.training_done::numeric / nullif(b.teachers, 0), 0)
    + 10 * least(b.lesson_plans::numeric / nullif(b.teachers, 0) * 2, 1)
    + 10 * (case when b.k12_synced then 1 else 0 end)
    )::int as kpi
  from base b
  order by b.name
$function$
;

-- ##### func equipment_availability(p_day date, p_period uuid)
CREATE OR REPLACE FUNCTION public.equipment_availability(p_day date, p_period uuid DEFAULT NULL::uuid)
 RETURNS TABLE(type_id uuid, usable integer, held integer, free integer)
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  select t.id,
         count(u.id) filter (where u.status not in ('broken', 'maintenance'))::int,
         coalesce(h.qty, 0)::int,
         greatest(count(u.id) filter (where u.status not in ('broken', 'maintenance')) - coalesce(h.qty, 0), 0)::int
    from equipment_types t
    left join equipment_units u on u.type_id = t.id and u.org_id = t.org_id
    left join (
      select (i->>'type_id')::uuid as type_id, sum(greatest(coalesce((i->>'qty')::int, 0), 0)) as qty
        from borrow_requests b, jsonb_array_elements(b.item_list) i
       where b.org_id = public.my_org_id() and b.status in ('approved', 'lent') and b.on_date = p_day
         and public.period_slot(b.on_date, b.period_id) && public.period_slot(p_day, p_period)
       group by 1) h on h.type_id = t.id
   where t.org_id = public.my_org_id() and public.is_staff()
   group by t.id, h.qty
$function$
;

-- ##### func equipment_shortage(p_org uuid, p_day date, p_period uuid, p_items jsonb, p_skip uuid)
CREATE OR REPLACE FUNCTION public.equipment_shortage(p_org uuid, p_day date, p_period uuid, p_items jsonb, p_skip uuid)
 RETURNS TABLE(type_id uuid, name text, need integer, usable integer, held integer)
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  with want as (
    select (i->>'type_id')::uuid as type_id, sum(greatest(coalesce((i->>'qty')::int, 0), 0))::int as qty
      from jsonb_array_elements(coalesce(p_items, '[]'::jsonb)) i where (i->>'type_id') is not null group by 1
  ), held as (
    select (i->>'type_id')::uuid as type_id, sum(greatest(coalesce((i->>'qty')::int, 0), 0))::int as qty
      from borrow_requests b, jsonb_array_elements(b.item_list) i
     where b.org_id = p_org and b.id is distinct from p_skip and b.status in ('approved', 'lent')
       and b.on_date = p_day and public.period_slot(b.on_date, b.period_id) && public.period_slot(p_day, p_period)
     group by 1
  )
  select w.type_id, t.name, w.qty,
         (select count(*) from equipment_units u where u.type_id = w.type_id and u.org_id = p_org and u.status not in ('broken', 'maintenance'))::int,
         coalesce(h.qty, 0)
    from want w join equipment_types t on t.id = w.type_id left join held h on h.type_id = w.type_id
$function$
;

-- ##### func generate_upcoming_sessions(p_days integer)
CREATE OR REPLACE FUNCTION public.generate_upcoming_sessions(p_days integer DEFAULT 28)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare v_today date := public.vn_today(); v_to date; s record; r jsonb; v_run bigint;
  n_sch int := 0; n_err int := 0; c int := 0; u int := 0; d int := 0; v_errs jsonb := '[]';
begin
  if p_days is null or p_days < 1 or p_days > 62 then raise exception 'Số ngày phải từ 1 đến 62' using errcode = '22023'; end if;
  v_to := v_today + p_days;
  insert into system_job_runs (job) values ('generate_upcoming_sessions') returning id into v_run;
  for s in select o.id, o.code from organizations o
            where o.type = 'school' and o.status = 'active' and public.org_has_module(o.id, 'timetable')
            order by o.code loop
    begin
      r := public.sync_teaching_sessions(s.id, v_today, v_to);
      n_sch := n_sch + 1;
      c := c + (r->>'created')::int; u := u + (r->>'updated')::int; d := d + (r->>'removed')::int;
    exception when others then                       -- một trường lỗi không chặn các trường khác
      n_err := n_err + 1;
      v_errs := v_errs || jsonb_build_object('school', s.code, 'error', sqlerrm);
    end;
  end loop;
  r := jsonb_build_object('from', v_today, 'to', v_to, 'schools', n_sch, 'failed', n_err,
                          'created', c, 'updated', u, 'removed', d, 'errors', v_errs);
  update system_job_runs set finished_at = now(), ok = (n_err = 0), result = r where id = v_run;
  delete from system_job_runs where job = 'generate_upcoming_sessions' and started_at < now() - interval '90 days';
  return r;
end $function$
;

-- ##### func holiday_resync()
CREATE OR REPLACE FUNCTION public.holiday_resync()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare v_today date := public.vn_today(); v_from date; v_to date; v_org uuid; s record;
begin
  -- Khoảng ngày bị ảnh hưởng: hợp của bản cũ và bản mới (sửa ngày nghỉ có thể dời khoảng)
  v_from := least(case when tg_op <> 'INSERT' then old.starts_on end, case when tg_op <> 'DELETE' then new.starts_on end);
  v_to   := greatest(case when tg_op <> 'INSERT' then old.ends_on end, case when tg_op <> 'DELETE' then new.ends_on end);
  v_org  := case when tg_op = 'DELETE' then old.org_id else new.org_id end;
  -- Chỉ trong 63 ngày tới (phạm vi tiết đã/sẽ được sinh); ngày nghỉ đã qua không đụng tiết cũ,
  -- ngày nghỉ xa hơn để lịch nền xử lý khi tới gần.
  v_from := greatest(v_from, v_today);
  v_to := least(v_to, v_today + 62);
  if v_to < v_from then return null; end if;
  for s in with recursive down as (
             select id, type, status from organizations where id = v_org
             union all select o.id, o.type, o.status from organizations o join down on o.parent_id = down.id)
           select id, code from down join organizations using (id)
            where down.type = 'school' and down.status = 'active' and public.org_has_module(id, 'timetable') loop
    begin
      perform public.sync_teaching_sessions(s.id, v_from, v_to);
    exception when others then                     -- lỗi của một trường không chặn việc lưu ngày nghỉ
      raise warning 'Ngày nghỉ: không đồng bộ được buổi dạy trường %: %', s.code, sqlerrm;
    end;
  end loop;
  return null;
end $function$
;

declare v_org uuid; v_class uuid;

  return new;
end $function$

-- ##### func library_adopt(p_item uuid, p_template boolean)
CREATE OR REPLACE FUNCTION public.library_adopt(p_item uuid, p_template boolean DEFAULT false)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  i library_items; v_org uuid := public.my_org_id(); v_teacher uuid := public.my_teacher_id();
  v_slides jsonb; v_plan uuid; v_first uuid; v_ids uuid[] := '{}'; v_lesson uuid; v_tpl boolean;
begin
  if v_org is null or (select type from organizations where id = v_org) <> 'school' then
    raise exception 'Chỉ trường học dùng được học liệu vào bài giảng' using errcode = '42501';
  end if;
  if not public.library_visible(p_item) then raise exception 'Không tìm thấy học liệu' using errcode = '42501'; end if;
  select * into i from library_items where id = p_item;
  if i.status <> 'published' and i.owner_org_id <> v_org then raise exception 'Học liệu chưa phát hành' using errcode = '42501'; end if;
  v_tpl := coalesce(p_template, false) or v_teacher is null;
  if v_tpl and not public.is_admin() then raise exception 'Chỉ quản trị trường tạo bài giảng mẫu của trường' using errcode = '42501'; end if;

  if i.kind = 'program' then
    -- Dùng cả chương trình: lấy mọi bài giảng được liên kết (mà đơn vị này thấy)
    for v_lesson in
      select distinct (l.value #>> '{}')::uuid
        from jsonb_array_elements(coalesce(i.content->'units', '[]')) u, jsonb_array_elements(coalesce(u->'lesson_ids', '[]')) l
    loop
      if public.library_visible(v_lesson) then
        v_ids := v_ids || ((public.library_adopt(v_lesson, v_tpl))->>'lesson_plan_id')::uuid;
      end if;
    end loop;
    insert into library_copies (item_id, org_id, adopted_version, lesson_plan_id) values (i.id, v_org, i.version, v_ids[1]);
    return jsonb_build_object('kind', 'program', 'lesson_plan_ids', to_jsonb(v_ids), 'count', coalesce(array_length(v_ids, 1), 0));
  end if;

  if i.kind = 'lesson' then
    v_slides := coalesce(i.content->'slides', '[]'::jsonb);
  else
    v_slides := jsonb_build_array(
      jsonb_build_object('id', 's1', 'kind', 'title', 'title', i.title, 'body', coalesce(i.summary, '')),
      case when i.content->>'type' in ('video', 'image')
        then jsonb_build_object('id', 's2', 'kind', 'media', 'title', i.title, 'url', coalesce(i.content->>'url', ''))
        else jsonb_build_object('id', 's2', 'kind', 'content', 'title', 'Tài liệu tham khảo', 'body', coalesce(i.content->>'url', '')) end);
  end if;
  insert into lesson_plans (org_id, teacher_id, title, subject, grades, program, is_template, status, summary, slides, library_item_id, library_version)
    values (v_org, case when v_tpl then null else v_teacher end, case when v_tpl then 'Mẫu: ' || i.title else i.title end,
            i.subject, i.grades, i.program, v_tpl, 'draft', i.summary, v_slides, i.id, i.version)
    returning id into v_plan;
  insert into library_copies (item_id, org_id, adopted_version, lesson_plan_id) values (i.id, v_org, i.version, v_plan);
  return jsonb_build_object('kind', i.kind, 'lesson_plan_id', v_plan);
end $function$
;

-- ##### func library_publish(p_item uuid, p_note text)
CREATE OR REPLACE FUNCTION public.library_publish(p_item uuid, p_note text DEFAULT NULL::text)
 RETURNS integer
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare i library_items; u record; v int;
begin
  select * into i from library_items where id = p_item for update;
  if i.id is null or i.owner_org_id is distinct from public.my_org_id() or not public.library_can_manage() then
    raise exception 'Bạn không phát hành được mục này' using errcode = '42501';
  end if;
  if i.kind = 'lesson' and jsonb_array_length(coalesce(i.content->'slides', '[]')) = 0 then raise exception 'Bài giảng chưa có slide nào'; end if;
  if i.kind = 'material' and coalesce(trim(i.content->>'url'), '') = '' then raise exception 'Học liệu chưa có đường dẫn'; end if;
  update library_items set status = 'published', version = version + 1, published_at = now(), change_note = nullif(trim(p_note), '')
   where id = i.id returning version into v;
  -- Báo người đã dùng bản cũ (một thông báo cho mỗi người, theo đơn vị)
  if v > 1 then
    for u in select distinct c.adopted_by, c.org_id from library_copies c where c.item_id = i.id and c.adopted_by is not null loop
      perform public.notify_user(u.adopted_by, u.org_id, 'library', 'Học liệu có bản cập nhật: ' || i.title,
        'Phiên bản ' || v || coalesce(' — ' || nullif(trim(p_note), ''), '') || '. Bản của bạn không bị thay đổi.',
        'Thư viện STEM', null, i.id, true);
    end loop;
  end if;
  return v;
end $function$
;

-- ##### func mark_teaching_session(p_session uuid, p_marks jsonb, p_note text)
CREATE OR REPLACE FUNCTION public.mark_teaching_session(p_session uuid, p_marks jsonb, p_note text DEFAULT NULL::text)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare v teaching_sessions; v_admin boolean; v_start timestamptz; v_bad int; v_today date := public.vn_today();
begin
  select * into v from teaching_sessions where id = p_session;
  if not found or not public.can_access_org(v.org_id) then raise exception 'Không tìm thấy tiết dạy' using errcode = '42501'; end if;
  v_admin := v.org_id = public.my_org_id() and public.has_perm('timetable', 'edit');
  if not v_admin and (v.teacher_id is null or v.teacher_id is distinct from public.my_teacher_id()) then
    raise exception 'Chỉ giáo viên dạy tiết này hoặc quản trị trường được điểm danh' using errcode = '42501';
  end if;
  if v.status = 'cancelled' then raise exception 'Tiết đã huỷ, không điểm danh' using errcode = '22023'; end if;
  if not v_admin then
    select (v.session_date + p.starts_at) at time zone 'Asia/Ho_Chi_Minh' into v_start from period_configs p where p.id = v.period_id;
    if now() < v_start - interval '10 minutes' then raise exception 'Chưa đến giờ tiết học' using errcode = '22023'; end if;
    if v.session_date < v_today - 7 then
      raise exception 'Quá 7 ngày, nhờ quản trị trường sửa điểm danh' using errcode = '22023';
    end if;
  end if;

  select count(*) into v_bad from jsonb_array_elements(coalesce(p_marks, '[]'::jsonb)) m
   where (m->>'status') is null or (m->>'status') not in ('present', 'absent', 'late', 'excused')
      or not exists (select 1 from students s where s.id = (m->>'student_id')::uuid and s.class_id = v.class_id);
  if v_bad > 0 then raise exception 'Danh sách điểm danh có % dòng không hợp lệ', v_bad using errcode = '22023'; end if;

  insert into lesson_attendance (session_id, student_id, org_id, status, note, marked_by, marked_at)
  select p_session, (m->>'student_id')::uuid, v.org_id, m->>'status', nullif(m->>'note', ''), auth.uid(), now()
    from jsonb_array_elements(coalesce(p_marks, '[]'::jsonb)) m
  on conflict (session_id, student_id) do update set
    status = excluded.status, note = excluded.note, marked_by = excluded.marked_by, marked_at = excluded.marked_at;

  update teaching_sessions set status = 'taught', note = coalesce(p_note, note), marked_by = auth.uid(), marked_at = now()
   where id = p_session;
  select * into v from teaching_sessions where id = p_session;
  return jsonb_build_object('present', v.present_count, 'absent', v.absent_count, 'late', v.late_count, 'excused', v.excused_count);
end $function$
;

-- ##### func my_permissions()
CREATE OR REPLACE FUNCTION public.my_permissions()
 RETURNS TABLE(module text, can_view boolean, can_detail boolean, can_create boolean, can_edit boolean, can_delete boolean)
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  with mine as (
    select m.role_id from memberships m where m.user_id = auth.uid() and m.org_id = public.my_org_id()
  ), mods as (
    select d.role_id, d.module from role_permissions d join mine using (role_id)
    union
    select o.role_id, o.module from org_role_permissions o join mine using (role_id) where o.org_id = public.my_org_id()
  )
  select m.module,
    bool_or(coalesce(o.can_view,   d.can_view,   false)),
    bool_or(coalesce(o.can_detail, d.can_detail, false)),
    bool_or(coalesce(o.can_create, d.can_create, false)),
    bool_or(coalesce(o.can_edit,   d.can_edit,   false)),
    bool_or(coalesce(o.can_delete, d.can_delete, false))
  from mods m
  left join org_role_permissions o on o.role_id = m.role_id and o.module = m.module and o.org_id = public.my_org_id()
  left join role_permissions d on d.role_id = m.role_id and d.module = m.module
  group by m.module
$function$
;

AS $function$
begin

-- ##### func org_entitlement(p_org uuid)
CREATE OR REPLACE FUNCTION public.org_entitlement(p_org uuid)
 RETURNS TABLE(state text, source_org uuid, contract_id uuid, contract_code text, plan_name text, modules text[], ends_on date)
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare v_depth int;
begin
  with recursive up as (
    select o.id, o.parent_id, 0 as d from organizations o where o.id = p_org
    union all
    select o.id, o.parent_id, u.d + 1 from organizations o join up u on o.id = u.parent_id where u.d < 10
  )
  select min(u.d) into v_depth from up u join contracts c on c.org_id = u.id
   where c.status <> 'draft' and (u.d = 0 or c.include_children);
  if v_depth is null then
    return query select 'open'::text, null::uuid, null::uuid, null::text, null::text, null::text[], null::date;
    return;
  end if;
  return query
  with recursive up as (
    select o.id, o.parent_id, 0 as d from organizations o where o.id = p_org
    union all
    select o.id, o.parent_id, u.d + 1 from organizations o join up u on o.id = u.parent_id where u.d < 10
  ), lvl as (
    select c.*, (c.status = 'active' and current_date >= c.starts_on and (c.ends_on is null or current_date <= c.ends_on)) as live
      from up u join contracts c on c.org_id = u.id
     where u.d = v_depth and c.status <> 'draft' and (u.d = 0 or c.include_children)
  ), pick as (
    select * from lvl order by live desc, starts_on desc limit 1
  )
  select case when k.live then 'active' else 'expired' end, k.org_id, k.id, k.code, p.name,
         case when k.live then p.modules else '{}'::text[] end, k.ends_on
    from pick k join service_plans p on p.id = k.plan_id;
end $function$
;

-- ##### func org_tree_stats()
CREATE OR REPLACE FUNCTION public.org_tree_stats()
 RETURNS TABLE(org_id uuid, members integer, schools integer, schools_active integer, teachers integer, students integer)
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  with recursive vis as (select t as id from public.my_org_tree() t where public.is_staff()),
  clos as (
    select id as anc, id as des from organizations where id in (select id from vis)
    union all
    select c.anc, o.id from clos c join organizations o on o.parent_id = c.des
  ),
  sch as (
    select o.id, o.activated_at is not null and o.status = 'active' as act,
      (select count(*) from teachers t where t.org_id = o.id and coalesce(t.status, 'active') = 'active') as tc,
      (select count(*) from students s where s.org_id = o.id and coalesce(s.status, 'studying') = 'studying') as sc
    from organizations o where o.type = 'school' and o.id in (select id from vis)
  ),
  mem as (select m.org_id, count(distinct m.user_id) as n from memberships m group by m.org_id)
  select v.id,
         coalesce((select n from mem where mem.org_id = v.id), 0)::int,
         count(s.id)::int, (count(s.id) filter (where s.act))::int,
         coalesce(sum(s.tc), 0)::int, coalesce(sum(s.sc), 0)::int
    from vis v left join clos c on c.anc = v.id left join sch s on s.id = c.des
   group by v.id
$function$
;

 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$

begin

-- ##### func platform_fulfill_order(p_order uuid, p_fulfillment text, p_catalog uuid, p_note text, p_value numeric)
CREATE OR REPLACE FUNCTION public.platform_fulfill_order(p_order uuid, p_fulfillment text, p_catalog uuid DEFAULT NULL::uuid, p_note text DEFAULT NULL::text, p_value numeric DEFAULT NULL::numeric)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare o package_orders; v_cat uuid; v_pkg uuid; v_label text;
begin
  perform public.require_platform();
  select * into o from package_orders where id = p_order for update;
  if o.id is null then raise exception 'Không tìm thấy đơn' using errcode = 'P0002'; end if;
  if p_fulfillment not in ('confirmed','delivering','delivered','cancelled') then raise exception 'Trạng thái không hợp lệ' using errcode = '23514'; end if;
  if o.status <> 'approved' and p_fulfillment <> 'cancelled' then raise exception 'Đơn chưa được duyệt theo quy trình của khách hàng' using errcode = '23514'; end if;
  if o.delivered_package_id is not null and p_fulfillment <> 'delivered' then raise exception 'Đơn đã giao, không đổi trạng thái được' using errcode = '23514'; end if;
  v_cat := coalesce(p_catalog, o.catalog_package_id);
  if v_cat is not null and not exists (select 1 from stem_packages where id = v_cat and org_id = public.platform_org_id()) then
    raise exception 'Gói không thuộc danh mục KDI' using errcode = '23514';
  end if;
  v_pkg := o.delivered_package_id;
  if p_fulfillment = 'delivered' and v_pkg is null then
    if v_cat is null then raise exception 'Chọn gói trong danh mục để giao' using errcode = '23514'; end if;
    v_pkg := public.copy_stem_package(v_cat, o.org_id, null, 'DH-' || left(o.id::text, 8));
  end if;
  update package_orders set fulfillment = p_fulfillment, catalog_package_id = v_cat,
         kdi_note = coalesce(nullif(trim(p_note), ''), kdi_note), quoted_value = coalesce(p_value, quoted_value),
         delivered_at = case when p_fulfillment = 'delivered' then coalesce(delivered_at, now()) else delivered_at end,
         delivered_package_id = v_pkg
   where id = o.id;
  v_label := case p_fulfillment when 'confirmed' then 'KDI đã xác nhận' when 'delivering' then 'KDI đang giao' when 'delivered' then 'KDI đã giao gói' else 'KDI huỷ đơn' end;
  if o.created_by is not null then
    insert into notifications (org_id, user_id, kind, title, body, page, portal, ref_id)
      values (o.org_id, o.created_by, 'package_order', v_label || ': ' || o.name, nullif(trim(p_note), ''),
              case when p_fulfillment = 'delivered' then 'Gói STEM' else 'Đề xuất mua gói' end, 'school_admin', o.id);
  end if;
  insert into audit_log (actor, org_id, entity, entity_id, action, summary, detail)
    values (auth.uid(), o.org_id, 'package_orders', o.id::text, 'update',
            v_label || ': ' || o.name || ' · ' || coalesce((select name from organizations where id = o.org_id), '?'),
            jsonb_build_object('fulfillment', p_fulfillment, 'catalog_package_id', v_cat, 'delivered_package_id', v_pkg));
  return v_pkg;
end $function$
;

-- ##### func platform_import_schools(p_rows jsonb)
CREATE OR REPLACE FUNCTION public.platform_import_schools(p_rows jsonb)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare r jsonb; v_parent uuid; v_level text; created int := 0; bad jsonb := '[]'::jsonb; v_msg text;
begin
  if not public.is_platform_admin() then raise exception 'Chỉ quản trị nền tảng' using errcode = '42501'; end if;
  if jsonb_typeof(p_rows) <> 'array' or jsonb_array_length(p_rows) > 2000 then raise exception 'Tối đa 2000 dòng một lần' using errcode = '22023'; end if;
  for r in select * from jsonb_array_elements(p_rows) loop
    begin
      if nullif(btrim(r->>'code'), '') is null or nullif(btrim(r->>'name'), '') is null then raise exception 'Thiếu mã hoặc tên trường'; end if;
      if exists (select 1 from organizations where code = btrim(r->>'code')) then raise exception 'Mã % đã có', btrim(r->>'code'); end if;
      select id into v_parent from organizations where code = btrim(r->>'parent_code') and type in ('department','district','province');
      if v_parent is null then raise exception 'Không tìm thấy Sở/Phòng có mã %', coalesce(r->>'parent_code', '(trống)'); end if;
      v_level := case lower(public.act_norm(r->>'level'))
        when 'mamnon' then 'mamnon' when 'mầm non' then 'mamnon' when 'tieuhoc' then 'tieuhoc' when 'tiểu học' then 'tieuhoc' when 'th' then 'tieuhoc'
        when 'thcs' then 'thcs' when 'trung học cơ sở' then 'thcs' when 'thpt' then 'thpt' when 'trung học phổ thông' then 'thpt'
        when '' then null else 'x' end;
      if v_level = 'x' then raise exception 'Cấp học "%" không hợp lệ (Mầm non / Tiểu học / THCS / THPT)', r->>'level'; end if;
      insert into organizations (type, parent_id, code, name, level, ward, address, contact_name, contact_phone, contact_email)
        values ('school', v_parent, btrim(r->>'code'), btrim(r->>'name'), v_level, nullif(btrim(r->>'ward'), ''), nullif(btrim(r->>'address'), ''),
                nullif(btrim(r->>'contact_name'), ''), nullif(btrim(r->>'contact_phone'), ''), nullif(btrim(r->>'contact_email'), ''));
      created := created + 1;
    exception when others then
      get stacked diagnostics v_msg = message_text;
      bad := bad || jsonb_build_object('line', r->'line', 'code', r->>'code', 'reason', v_msg);
    end;
  end loop;
  return jsonb_build_object('created', created, 'skipped', bad);
end $function$
;

-- ##### func platform_overview()
CREATE OR REPLACE FUNCTION public.platform_overview()
 RETURNS jsonb
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare res jsonb; checks jsonb := '[]'::jsonb;
  v_items jsonb; v_n int;
begin
  if not public.is_platform_admin() then raise exception 'Chỉ quản trị nền tảng' using errcode = '42501'; end if;

  -- 1. Đơn vị sai cấp
  select count(*), coalesce(jsonb_agg(o.name order by o.name), '[]') into v_n, v_items
    from organizations o left join organizations p on p.id = o.parent_id
   where not public.org_parent_ok(o.type, p.type);
  checks := checks || jsonb_build_object('level', 'bad', 'key', 'wrong_parent', 'title', 'Đơn vị đặt sai cấp (vd: trường nằm trực tiếp dưới Bộ)', 'count', v_n, 'items', v_items);
  -- 2. Sở chưa thuộc tỉnh
  select count(*), coalesce(jsonb_agg(name order by name), '[]') into v_n, v_items
    from organizations where type = 'department' and parent_id is null;
  checks := checks || jsonb_build_object('level', 'warn', 'key', 'dept_no_province', 'title', 'Sở chưa gắn vào Tỉnh', 'count', v_n, 'items', v_items);
  -- 3. Đơn vị chưa có ai phụ trách
  select count(*), coalesce(jsonb_agg(name order by name), '[]') into v_n, v_items
    from organizations o where o.type in ('province','department','district','school') and o.status = 'active'
     and not exists (select 1 from memberships m join roles r on r.id = m.role_id
                      where m.org_id = o.id and r.key <> 'student');
  checks := checks || jsonb_build_object('level', 'warn', 'key', 'no_staff', 'title', 'Đơn vị chưa có tài khoản cán bộ nào', 'count', v_n, 'items', v_items);
  -- 4. Tài khoản đang trỏ tới đơn vị mà họ không có vai trò
  select count(*), coalesce(jsonb_agg(coalesce(p.full_name, p.email) order by p.full_name), '[]') into v_n, v_items
    from profiles p where p.org_id is not null
     and not exists (select 1 from memberships m where m.user_id = p.id and m.org_id = p.org_id);
  checks := checks || jsonb_build_object('level', 'bad', 'key', 'stale_active_org', 'title', 'Tài khoản trỏ tới đơn vị không còn vai trò', 'count', v_n, 'items', v_items);
  -- 5. Tài khoản chưa thuộc đơn vị nào
  select count(*), coalesce(jsonb_agg(coalesce(p.full_name, p.email) order by p.full_name), '[]') into v_n, v_items
    from profiles p where not exists (select 1 from memberships m where m.user_id = p.id);
  checks := checks || jsonb_build_object('level', 'info', 'key', 'no_membership', 'title', 'Tài khoản chưa thuộc đơn vị nào (đang chờ cấp quyền)', 'count', v_n, 'items', v_items);
  -- 6. Trường chưa kích hoạt / đơn vị tạm dừng
  select count(*), coalesce(jsonb_agg(name order by name), '[]') into v_n, v_items
    from organizations where type = 'school' and activated_at is null;
  checks := checks || jsonb_build_object('level', 'info', 'key', 'not_activated', 'title', 'Trường chưa kích hoạt', 'count', v_n, 'items', v_items);
  select count(*), coalesce(jsonb_agg(name order by name), '[]') into v_n, v_items
    from organizations where status = 'suspended';
  checks := checks || jsonb_build_object('level', 'info', 'key', 'suspended', 'title', 'Đơn vị đang tạm dừng', 'count', v_n, 'items', v_items);

  select jsonb_build_object(
    'org_counts', (select coalesce(jsonb_object_agg(type, n), '{}') from (select type, count(*) n from organizations group by type) x),
    'schools_active', (select count(*) from organizations where type = 'school' and activated_at is not null and status = 'active'),
    'users', (select count(*) from profiles),
    'users_active_7d', (select count(distinct actor) from audit_log where at > now() - interval '7 days'),
    'memberships', (select count(*) from memberships),
    'by_role', (select coalesce(jsonb_agg(jsonb_build_object('name', name, 'n', n) order by n desc), '[]')
                  from (select r.name, count(*) n from memberships m join roles r on r.id = m.role_id group by r.name) x),
    'invitations_pending', (select count(*) from invitations where status = 'pending'),
    'teachers', (select count(*) from teachers where coalesce(status, 'active') = 'active'),
    'students', (select count(*) from students where coalesce(status, 'studying') = 'studying'),
    'checks', checks,
    'recent', (select coalesce(jsonb_agg(x order by x.at desc), '[]') from (
                 select a.at, a.summary, a.entity, coalesce(p.full_name, p.email, 'Hệ thống') as actor
                   from audit_log a left join profiles p on p.id = a.actor order by a.at desc limit 8) x)
  ) into res;
  return res;
end $function$
;

-- ##### func platform_school_registry()
CREATE OR REPLACE FUNCTION public.platform_school_registry()
 RETURNS TABLE(org_id uuid, code text, name text, level text, ward text, address text, parent_id uuid, parent_name text, province_id uuid, province_name text, status text, activated boolean, customer_status text, contact_name text, contact_phone text, contact_email text, kdi_owner uuid, kdi_owner_name text, go_live_on date, created_at timestamp with time zone, ent_state text, plan_name text, contract_code text, contract_ends_on date, contract_source text, users integer, admins integer, pending_invites integer, last_login timestamp with time zone, last_activity timestamp with time zone, active_teachers_7d integer, planned_4w integer, taught_4w integer, taught_14d integer, rollout_done integer, rollout_total integer, kickoff_at timestamp with time zone, accepted_at timestamp with time zone, open_tickets integer, urgent_tickets integer, stage text, health text, reasons text[], stats jsonb)
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
#variable_conflict use_column
declare v_today date := public.vn_today();
begin
  if not public.is_platform_admin() then raise exception 'Chỉ quản trị nền tảng' using errcode = '42501'; end if;

  return query
  with st as (select * from public.dept_school_stats()),
  sc as (
    select o.*, par.name as par_name,
      public.org_ancestor_of_type(o.id, 'province') as prov_id,
      coalesce(o.customer_status, par.customer_status, gp.customer_status) as eff_cs
    from organizations o left join organizations par on par.id = o.parent_id left join organizations gp on gp.id = par.parent_id
    where o.type = 'school'
  ),
  ent as (select sc.id, e.* from sc cross join lateral public.org_entitlement(sc.id) e),
  chk as (select c.org_id, count(*) filter (where c.step in ('kickoff','admin_training','handover','teacher_training','acceptance'))::int as manual,
                 max(c.done_at) filter (where c.step = 'kickoff') as kickoff_at, max(c.done_at) filter (where c.step = 'acceptance') as accepted_at
            from rollout_checks c group by c.org_id),
  mem as (select m.org_id, count(distinct m.user_id)::int as users,
                 count(distinct m.user_id) filter (where r.key in ('principal','it_admin') and r.org_id is null)::int as admins
            from memberships m join roles r on r.id = m.role_id group by m.org_id),
  inv as (select i.org_id, count(*)::int as n from invitations i where i.status = 'pending' group by i.org_id),
  ses as (
    select ts.org_id,
      count(*) filter (where ts.status <> 'cancelled' and ts.session_date between v_today - 27 and v_today
                         and (ts.session_date + p.starts_at) at time zone 'Asia/Ho_Chi_Minh' <= now())::int as planned,
      count(*) filter (where ts.status = 'taught' and ts.session_date between v_today - 27 and v_today)::int as taught,
      count(*) filter (where ts.status = 'taught' and ts.session_date between v_today - 13 and v_today)::int as taught14,
      count(distinct ts.teacher_id) filter (where ts.status = 'taught' and ts.marked_at >= now() - interval '7 days')::int as act7
    from teaching_sessions ts join period_configs p on p.id = ts.period_id
    where ts.session_date >= v_today - 27 group by ts.org_id),
  tk as (select t.org_id, count(*) filter (where t.status not in ('resolved','closed'))::int as open_n,
                count(*) filter (where t.status not in ('resolved','closed') and t.priority = 'urgent')::int as urgent_n
           from support_tickets t group by t.org_id),
  base as (
    select sc.*, coalesce(st.activated, false) as activated, st.total_periods, st.last_activity, ent.state as e_state, ent.plan_name as e_plan, ent.contract_code as e_code, ent.ends_on as e_ends,
      (select name from organizations where id = ent.source_org) as e_src,
      coalesce(mem.users, 0) as n_users, coalesce(mem.admins, 0) as n_admins, coalesce(inv.n, 0) as n_inv,
      lg.last_login as ll, coalesce(ses.planned, 0) as s_planned, coalesce(ses.taught, 0) as s_taught,
      coalesce(ses.taught14, 0) as s_taught14, coalesce(ses.act7, 0) as s_act7,
      coalesce(tk.open_n, 0) as t_open, coalesce(tk.urgent_n, 0) as t_urgent,
      chk.kickoff_at as k_at, chk.accepted_at as a_at,
      coalesce(chk.manual, 0)
        + (ent.state = 'active')::int + coalesce(st.activated, false)::int + coalesce(st.has_year, false)::int
        + (coalesce(st.teachers, 0) > 0)::int + (coalesce(st.students, 0) > 0)::int + (coalesce(st.rooms_active, 0) > 0)::int
        + (coalesce(st.units, 0) > 0)::int + (coalesce(st.total_periods, 0) > 0)::int
        + (st.last_activity >= now() - interval '7 days')::int as r_done,
      to_jsonb(st) as st_json
    from sc
    left join st on st.school_id = sc.id
    left join ent on ent.id = sc.id
    left join chk on chk.org_id = sc.id
    left join mem on mem.org_id = sc.id
    left join inv on inv.org_id = sc.id
    left join public.school_last_login() lg on lg.org_id = sc.id
    left join ses on ses.org_id = sc.id
    left join tk on tk.org_id = sc.id
  ),
  staged as (
    select b.*,
      case when b.status = 'suspended' or b.eff_cs = 'paused' then 'paused'
           when b.eff_cs = 'ended' then 'ended'
           when b.eff_cs in ('prospect','negotiating') and b.e_state is distinct from 'active' then 'prospect'
           when b.a_at is not null or (b.go_live_on is not null and b.go_live_on <= v_today) or b.s_taught14 > 0 then 'operating'
           when b.k_at is not null or b.activated or b.r_done >= 3 then 'rollout'
           else 'setup' end as stg,
      case when b.e_ends is null then null else b.e_ends - v_today end as days_left
    from base b
  ),
  why as (
    select s.*,
      array_remove(array[
        case when s.e_state = 'expired' then 'R:Hợp đồng đã hết hạn' end,
        case when s.e_state = 'active' and s.days_left between 0 and 30 then 'R:Hợp đồng còn ' || s.days_left || ' ngày' end,
        case when s.stg = 'operating' and coalesce(s.total_periods, 0) > 0 and s.s_taught14 = 0 then 'R:Không có tiết nào được điểm danh trong 14 ngày' end,
        case when s.t_urgent > 0 then 'R:Có ' || s.t_urgent || ' phiếu hỗ trợ khẩn chưa xử lý' end,
        case when s.e_state = 'active' and s.days_left between 31 and 60 then 'W:Hợp đồng còn ' || s.days_left || ' ngày' end,
        case when s.stg = 'operating' and s.s_planned >= 10 and s.s_taught * 2 < s.s_planned then 'W:Điểm danh dưới 50% số tiết (4 tuần)' end,
        case when s.stg = 'rollout' and s.k_at < now() - interval '60 days' then 'W:Triển khai quá 60 ngày chưa nghiệm thu' end,
        case when s.stg in ('rollout','operating') and s.n_admins = 0 then 'W:Chưa có quản trị trường' end,
        case when s.stg in ('rollout','operating') and s.n_users > 0 and (s.ll is null or s.ll < now() - interval '14 days')
             then 'W:Không ai đăng nhập trong 14 ngày' end
      ], null) as rs
    from staged s
  )
  select w.id, w.code, w.name, w.level, w.ward, w.address, w.parent_id, w.par_name, w.prov_id,
    (select name from organizations where id = w.prov_id),
    w.status, coalesce(w.activated, false), w.eff_cs, w.contact_name, w.contact_phone, w.contact_email,
    w.kdi_owner, (select full_name from profiles where id = w.kdi_owner), w.go_live_on, w.created_at,
    coalesce(w.e_state, 'open'), w.e_plan, w.e_code, w.e_ends, w.e_src,
    w.n_users, w.n_admins, w.n_inv, w.ll, w.last_activity,
    w.s_act7, w.s_planned, w.s_taught, w.s_taught14,
    w.r_done, 14, w.k_at, w.a_at, w.t_open, w.t_urgent,
    w.stg,
    case when w.stg in ('paused','ended','prospect') then 'na'
         when exists (select 1 from unnest(w.rs) x where x like 'R:%') then 'risk'
         when cardinality(w.rs) > 0 then 'watch' else 'good' end,
    array(select substr(x, 3) from unnest(w.rs) x order by x),
    w.st_json
  from why w
  order by w.name;
end $function$
;
-- ##### func platform_school_trend(p_org uuid)
CREATE OR REPLACE FUNCTION public.platform_school_trend(p_org uuid)
 RETURNS TABLE(week_start date, planned integer, taught integer, cancelled integer, present integer, absent integer, active_teachers integer)
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
#variable_conflict use_column
declare v_mon date := date_trunc('week', public.vn_today())::date;
begin
  if not public.is_platform_admin() then raise exception 'Chỉ quản trị nền tảng' using errcode = '42501'; end if;

  select w::date,
    count(ts.id) filter (where ts.status <> 'cancelled' and (ts.session_date + p.starts_at) at time zone 'Asia/Ho_Chi_Minh' <= now())::int,
    count(ts.id) filter (where ts.status = 'taught')::int,
    count(ts.id) filter (where ts.status = 'cancelled')::int,
    coalesce(sum(ts.present_count + ts.late_count), 0)::int, coalesce(sum(ts.absent_count), 0)::int,
    count(distinct ts.teacher_id) filter (where ts.status = 'taught')::int
  from generate_series(v_mon - 49, v_mon, interval '7 days') w
  left join teaching_sessions ts on ts.org_id = p_org and ts.session_date between w::date and w::date + 6
  left join period_configs p on p.id = ts.period_id
  group by w order by w;
end $function$
;
-- ##### func platform_security_overview()
CREATE OR REPLACE FUNCTION public.platform_security_overview()
 RETURNS jsonb
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v jsonb; v_sessions jsonb; v_factors jsonb; v_note text;
  v_roles text[] := (select mfa_roles from security_settings where id = 1);
begin
  if not public.is_platform_admin() then raise exception 'Chỉ quản trị nền tảng' using errcode = '42501'; end if;
  begin
    if to_regclass('auth.sessions') is not null then
      execute $q$select coalesce(jsonb_agg(x order by x->>'updated_at' desc nulls last), '[]'::jsonb) from (
        select to_jsonb(s) - 'factor_id' - 'tag' || jsonb_build_object('email', p.email, 'full_name', p.full_name) as x
          from auth.sessions s left join public.profiles p on p.id = s.user_id
         order by coalesce(s.updated_at, s.created_at) desc limit 300) t$q$ into v_sessions;
    end if;
    if to_regclass('auth.mfa_factors') is not null then
      execute $q$select coalesce(jsonb_agg(distinct user_id), '[]'::jsonb) from auth.mfa_factors where status = 'verified'$q$ into v_factors;
    end if;
  exception when insufficient_privilege then
    v_note := 'Chưa đọc được dữ liệu phiên/2FA của Supabase Auth (thiếu quyền trên schema auth).';
  end;
  select jsonb_build_object(
    'settings', (select to_jsonb(s) from security_settings s where id = 1),
    'users', (select count(*) from profiles),
    'locked', (select coalesce(jsonb_agg(jsonb_build_object('id', p.id, 'email', p.email, 'full_name', p.full_name, 'locked_at', p.locked_at,
                 'reason', p.locked_reason, 'by', (select coalesce(full_name, email) from profiles b where b.id = p.locked_by)) order by p.locked_at desc), '[]'::jsonb)
                 from profiles p where p.locked_at is not null),
    'mfa_users', (select coalesce(jsonb_agg(jsonb_build_object('id', p.id, 'email', p.email, 'full_name', p.full_name,
                    'roles', (select jsonb_agg(distinct r.name) from memberships m join roles r on r.id = m.role_id where m.user_id = p.id and r.key = any(v_roles)),
                    'enrolled', case when v_factors is null then null else v_factors ? p.id::text end) order by p.full_name), '[]'::jsonb)
                   from profiles p where exists (select 1 from memberships m join roles r on r.id = m.role_id where m.user_id = p.id and r.key = any(v_roles))),
    'sessions', v_sessions,
    'note', v_note) into v;
  return v;
end $function$

-- ##### func room_overview(p_from date, p_to date)
CREATE OR REPLACE FUNCTION public.room_overview(p_from date, p_to date)
 RETURNS TABLE(room_id uuid, org_id uuid, school text, parent_name text, room text, code text, status text, capacity integer, tier text, slots integer, booked integer, from_sessions integer, from_registrations integer, from_borrows integer, conflicts integer, tkb_periods integer)
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare v_range tstzrange;
begin
  if not public.is_staff() then raise exception 'Không có quyền' using errcode = '42501'; end if;
  if p_from is null or p_to is null or p_to < p_from or p_to - p_from > 400 then
    raise exception 'Khoảng ngày không hợp lệ' using errcode = '22023';
  end if;
  v_range := tstzrange(p_from::timestamp at time zone 'Asia/Ho_Chi_Minh', (p_to + 1)::timestamp at time zone 'Asia/Ho_Chi_Minh', '[)');
  return query
  with rm as (
    select r.*, o.name as school, par.name as parent_name from stem_rooms r
      join organizations o on o.id = r.org_id left join organizations par on par.id = o.parent_id
     where r.org_id in (select public.my_org_tree())
  ), days as (
    select count(*)::int as n from generate_series(p_from, p_to, interval '1 day') g where extract(isodow from g) between 1 and 6
  ), bk as (
    select b.room_id, count(*) as n, count(*) filter (where b.source = 'teaching_session') as s,
           count(*) filter (where b.source = 'stem_registration') as r, count(*) filter (where b.source = 'borrow_request') as w
      from resource_bookings b where b.room_id in (select id from rm) and b.during && v_range group by b.room_id
  ), cf as (
    select c.room_id, count(*) as n from booking_conflicts c where c.room_id in (select id from rm) and c.during && v_range group by c.room_id
  )
  select rm.id, rm.org_id, rm.school, rm.parent_name, rm.name, rm.code, rm.status, rm.capacity, rm.tier,
    ((select n from days) * (select count(*) from period_configs p where p.org_id = rm.org_id))::int,
    coalesce(bk.n, 0)::int, coalesce(bk.s, 0)::int, coalesce(bk.r, 0)::int, coalesce(bk.w, 0)::int, coalesce(cf.n, 0)::int,
    (select count(*) from timetable_entries e where e.room_id = rm.id)::int
  from rm left join bk on bk.room_id = rm.id left join cf on cf.room_id = rm.id
  order by rm.school, rm.name;
end $function$
;

-- ##### func self_study_snapshot()
CREATE OR REPLACE FUNCTION public.self_study_snapshot()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare v_kind text; v_content jsonb; v_org uuid; v_status text; v_owner uuid; v_tpl boolean; v_err text;
begin
  select org_id into v_org from classes where id = new.class_id;
  new.org_id := v_org;
  if new.kind = 'activity' then
    if new.activity_id is null then raise exception 'Chọn hoạt động cần giao' using errcode = '23514'; end if;
    select kind, content, org_id, status, teacher_id, is_template into v_kind, v_content, v_org, v_status, v_owner, v_tpl
      from activities where id = new.activity_id;
    if v_org is distinct from new.org_id then raise exception 'Hoạt động không thuộc trường này' using errcode = '42501'; end if;
    if v_status <> 'published' and not v_tpl and v_owner is distinct from new.teacher_id then
      raise exception 'Chỉ giao được hoạt động đã xuất bản hoặc của chính mình' using errcode = '42501'; end if;
    v_err := public.activity_problem(v_kind, v_content);
    if v_err is not null then raise exception 'Hoạt động chưa hoàn chỉnh: %', v_err using errcode = '23514'; end if;
    new.activity_kind := v_kind;
  elsif new.kind = 'lesson' then
    if new.lesson_plan_id is null then raise exception 'Chọn bài giảng cần giao' using errcode = '23514'; end if;
    select 'lesson', jsonb_build_object('slides', slides), org_id into v_kind, v_content, v_org from lesson_plans where id = new.lesson_plan_id;
    if v_org is distinct from new.org_id then raise exception 'Bài giảng không thuộc trường này' using errcode = '42501'; end if;
    new.activity_kind := null;
  else
    v_kind := 'link'; v_content := jsonb_build_object('url', new.url); new.activity_kind := null;
  end if;
  if tg_op = 'INSERT' or new.activity_id is distinct from old.activity_id or new.lesson_plan_id is distinct from old.lesson_plan_id
     or new.url is distinct from old.url or new.kind is distinct from old.kind then
    -- Bảng nội dung có khoá ngoại tới item → ghi sau khi item đã có (trigger AFTER bên dưới)
    perform set_config('app.ss_kind', v_kind, true);
    perform set_config('app.ss_content', v_content::text, true);
  else
    perform set_config('app.ss_kind', '', true);
  end if;
  return new;
end $function$
;

-- ##### func self_study_submit(p_item uuid, p_answers jsonb)
CREATE OR REPLACE FUNCTION public.self_study_submit(p_item uuid, p_answers jsonb DEFAULT NULL::jsonb)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare i self_study_items; c self_study_content; p self_study_progress; v_st uuid := public.my_student_id(); g jsonb; v_score numeric;
begin
  if not public.is_student() or v_st is null then raise exception 'Chỉ học sinh làm bài tự học' using errcode = '42501'; end if;
  select * into i from self_study_items where id = p_item and class_id = public.my_class_id() and status <> 'draft' and opens_at <= now();
  if not found then raise exception 'Không tìm thấy bài tự học của lớp em' using errcode = '42501'; end if;
  if i.status = 'closed' then raise exception 'Thầy cô đã khoá bài này' using errcode = '22023'; end if;
  select * into c from self_study_content where item_id = i.id;
  insert into self_study_progress (item_id, student_id, org_id) values (i.id, v_st, i.org_id) on conflict do nothing;
  select * into p from self_study_progress where item_id = i.id and student_id = v_st for update;
  if i.max_attempts > 0 and p.attempts >= i.max_attempts then raise exception 'Em đã hết lượt làm bài này' using errcode = '22023'; end if;
  if p_answers is not null and length(p_answers::text) > 20000 then raise exception 'Câu trả lời quá dài' using errcode = '22023'; end if;
  g := public.activity_grade(c.kind, c.content, p_answers);
  v_score := case when (g->>'total')::int > 0 then round(10.0 * (g->>'correct')::int / (g->>'total')::int, 1) end;
  update self_study_progress set status = 'done', attempts = attempts + 1, answers = p_answers,
    score = v_score, best_score = case when v_score is null then best_score else greatest(coalesce(best_score, 0), v_score) end,
    correct = case when (g->>'total')::int > 0 then (g->>'correct')::int end, total = case when (g->>'total')::int > 0 then (g->>'total')::int end,
    late = late or (i.due_at is not null and now() > i.due_at),
    submitted_at = now(), completed_at = coalesce(completed_at, now())
  where item_id = i.id and student_id = v_st
  returning * into p;
  return jsonb_build_object('score', p.score, 'best_score', p.best_score, 'correct', p.correct, 'total', p.total,
    'grade', g, 'attempts', p.attempts,
    'attempts_left', case when i.max_attempts = 0 then null else greatest(0, i.max_attempts - p.attempts) end,
    'solution', case when public.self_study_reveal(i, p) then c.content end);
end $function$
;

-- ##### func start_class_session(p_class uuid, p_title text, p_lesson uuid, p_plan uuid)
CREATE OR REPLACE FUNCTION public.start_class_session(p_class uuid, p_title text, p_lesson uuid DEFAULT NULL::uuid, p_plan uuid DEFAULT NULL::uuid)
 RETURNS class_sessions
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$

        returning * into s;
      return s;

-- ##### func support_ticket_before()

 RETURNS trigger
 LANGUAGE plpgsql

 SET search_path TO 'public'

  if tg_op = 'INSERT' then
    new.code := 'HT-' || lpad(nextval('support_ticket_seq')::text, 5, '0');
    new.created_at := now();
    new.due_at := now() + public.support_sla(new.priority);
    if not v_platform then new.status := 'new'; new.assignee := null; new.reported_by := auth.uid(); end if;
  else
    if not v_platform and current_setting('role', true) = 'authenticated' then
      -- Khách hàng chỉ đóng phiếu hoặc mở lại; không đổi người xử lý, mức ưu tiên, hạn, đơn vị
      if (new.org_id, new.category, new.priority, new.assignee, new.due_at, new.reported_by, new.subject, new.warranty_ticket_id, new.first_response_at)
         is distinct from (old.org_id, old.category, old.priority, old.assignee, old.due_at, old.reported_by, old.subject, old.warranty_ticket_id, old.first_response_at)
         or (new.status is distinct from old.status and new.status not in ('closed','in_progress')) then

    end if;
    if new.status in ('resolved','closed') and old.status not in ('resolved','closed') then new.resolved_at := now();
    elsif new.status not in ('resolved','closed') then new.resolved_at := null; end if;

-- ##### func sync_teaching_sessions(p_org uuid, p_from date, p_to date)
CREATE OR REPLACE FUNCTION public.sync_teaching_sessions(p_org uuid, p_from date, p_to date)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare v_ins int; v_upd int; v_del int; v_today date := public.vn_today(); v_skip int;
begin
  if p_org is null or (select type from organizations where id = p_org) is distinct from 'school' then
    raise exception 'Chỉ sinh buổi dạy cho trường học' using errcode = '22023';
  end if;
  if p_from is null or p_to is null or p_to < p_from or p_to - p_from > 62 then
    raise exception 'Khoảng ngày không hợp lệ (tối đa 63 ngày)' using errcode = '22023';
  end if;

  if to_regclass('pg_temp._ts_src') is null then      -- gọi nhiều lần trong một giao dịch (lịch nền) → tạo một lần
    create temp table _ts_src (entry_id uuid, class_id uuid, teacher_id uuid, period_id uuid, room_id uuid,
      day date, year_id uuid, subject text, kind text) on commit drop;
  end if;
  truncate _ts_src;
  insert into _ts_src
  select distinct on (e.class_id, d.day, e.period_id)
         e.id, e.class_id, e.teacher_id, e.period_id, e.room_id, d.day, d.year_id, e.subject, e.kind
  from (select g::date as day, y.id as year_id
          from generate_series(p_from, p_to, interval '1 day') g
          join academic_years y on y.org_id = p_org and g::date between y.starts_on and y.ends_on
         where extract(isodow from g) between 1 and 6
           and public.is_holiday(p_org, g::date) is null) d          -- ngày nghỉ của trường và mọi cấp trên
  join timetable_entries e on e.org_id = p_org
   and e.weekday = extract(isodow from d.day)::int + 1
   and (e.year_id is null or e.year_id = d.year_id)
  order by e.class_id, d.day, e.period_id, (e.year_id is null), e.id;   -- ưu tiên TKB riêng của học kỳ

  -- Tiết tương lai chưa điểm danh không còn trong nguồn (đổi TKB, rơi vào ngày nghỉ) → xoá
  delete from teaching_sessions s
   where s.org_id = p_org and s.session_date between greatest(p_from, v_today) and p_to
     and s.status = 'scheduled' and not s.is_makeup
     and not exists (select 1 from lesson_attendance a where a.session_id = s.id)
     and not exists (select 1 from _ts_src x where x.class_id = s.class_id and x.day = s.session_date and x.period_id = s.period_id);
  get diagnostics v_del = row_count;

  with up as (
    insert into teaching_sessions (org_id, year_id, timetable_entry_id, class_id, teacher_id, period_id, room_id, session_date, subject, kind)
    select p_org, year_id, entry_id, class_id, teacher_id, period_id, room_id, day, subject, kind from _ts_src
    on conflict (class_id, session_date, period_id) do update set
      year_id = excluded.year_id, timetable_entry_id = excluded.timetable_entry_id, teacher_id = excluded.teacher_id,
      room_id = excluded.room_id, subject = excluded.subject, kind = excluded.kind
    where teaching_sessions.status = 'scheduled' and not teaching_sessions.is_makeup and teaching_sessions.session_date >= v_today
      and (teaching_sessions.timetable_entry_id, teaching_sessions.teacher_id, teaching_sessions.room_id, teaching_sessions.subject, teaching_sessions.kind)
          is distinct from (excluded.timetable_entry_id, excluded.teacher_id, excluded.room_id, excluded.subject, excluded.kind)
    returning (xmax = 0) as inserted
  ) select count(*) filter (where inserted), count(*) filter (where not inserted) into v_ins, v_upd from up;

  select count(*) into v_skip from generate_series(p_from, p_to, interval '1 day') g
   where extract(isodow from g) between 1 and 6 and public.is_holiday(p_org, g::date) is not null;

  return jsonb_build_object('created', v_ins, 'updated', v_upd, 'removed', v_del, 'holidays', v_skip);
end $function$
;

-- ##### func teacher_slot_guard()
CREATE OR REPLACE FUNCTION public.teacher_slot_guard()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare v_busy text; v_name text;
begin
  if coalesce(current_setting('kdi.booking_soft', true), '') = 'on' then return new; end if;
  if tg_table_name = 'teaching_sessions' then
    if new.teacher_id is null or new.status = 'cancelled' then return new; end if;
    if tg_op = 'UPDATE' and (old.teacher_id, old.session_date, old.period_id, old.class_id, old.status = 'cancelled')
       is not distinct from (new.teacher_id, new.session_date, new.period_id, new.class_id, new.status = 'cancelled') then return new; end if;
    -- Tiết sinh từ TKB, đúng GV của TKB: TKB đã chặn trùng
    if new.timetable_entry_id is not null and not new.is_makeup
       and new.teacher_id is not distinct from (select e.teacher_id from timetable_entries e where e.id = new.timetable_entry_id) then return new; end if;
    v_busy := public.teacher_slot_busy(new.teacher_id, new.session_date, new.period_id, new.class_id, new.id, null);
  else
    if new.status not in ('pending', 'approved') then return new; end if;
    if tg_op = 'UPDATE' and (old.teacher_id, old.on_date, old.period_id, old.class_id, old.status in ('pending', 'approved'))
       is not distinct from (new.teacher_id, new.on_date, new.period_id, new.class_id, true) then return new; end if;
    v_busy := public.teacher_slot_busy(new.teacher_id, new.on_date, new.period_id, new.class_id, null, new.id);
  end if;
  if v_busy is not null then
    select full_name into v_name from teachers where id = new.teacher_id;
    raise exception 'Giáo viên % đã có tiết trùng giờ: %', coalesce(v_name, ''), v_busy
      using errcode = '23P01', hint = 'Chọn tiết khác hoặc đổi giáo viên.';
  end if;
  return new;
end $function$
;

-- ##### func timetable_conflicts()
CREATE OR REPLACE FUNCTION public.timetable_conflicts()
 RETURNS TABLE(org_id uuid, kind text, weekday integer, period_name text, resource text, classes text)
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  with e as (
    select e.*, p.name as period_name from timetable_entries e join period_configs p on p.id = e.period_id
    where e.org_id in (select public.my_org_tree()) and public.is_staff()
  )
  select a.org_id, 'teacher', a.weekday, a.period_name, t.full_name, string_agg(distinct c.name, ', ')
  from e a join e b on b.org_id = a.org_id and b.weekday = a.weekday and b.period_id = a.period_id
       and b.teacher_id = a.teacher_id and b.class_id <> a.class_id
       and (a.year_id is null or b.year_id is null or a.year_id = b.year_id)
  join teachers t on t.id = a.teacher_id join classes c on c.id in (a.class_id, b.class_id)
  group by a.org_id, a.weekday, a.period_name, t.full_name
  union all
  select a.org_id, 'room', a.weekday, a.period_name, r.name, string_agg(distinct c.name, ', ')
  from e a join e b on b.org_id = a.org_id and b.weekday = a.weekday and b.period_id = a.period_id
       and b.room_id = a.room_id and b.class_id <> a.class_id
       and (a.year_id is null or b.year_id is null or a.year_id = b.year_id)
  join stem_rooms r on r.id = a.room_id join classes c on c.id in (a.class_id, b.class_id)
  group by a.org_id, a.weekday, a.period_name, r.name
$function$
;

 RETURNS trigger

 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$

begin

     is not distinct from (new.room_id, new.teacher_id, new.weekday, new.period_id, new.year_id, new.class_id) then return new; end if;

       and e.room_id = new.room_id and e.class_id <> new.class_id
       and (e.year_id is null or new.year_id is null or e.year_id = new.year_id)
     limit 1;
    if v is not null then raise exception 'Trùng phòng: % ở cùng thứ, cùng tiết', v using errcode = '23P01'; end if;
  end if;
  if new.teacher_id is not null then

       and e.teacher_id = new.teacher_id and e.class_id <> new.class_id
       and (e.year_id is null or new.year_id is null or e.year_id = new.year_id)
     limit 1;
    if v is not null then raise exception 'Trùng giáo viên: % ở cùng thứ, cùng tiết', v using errcode = '23P01'; end if;
  end if;
  return new;
end $function$
;
-- ##### func timetable_overview()
CREATE OR REPLACE FUNCTION public.timetable_overview()
 RETURNS TABLE(org_id uuid, school text, parent_name text, classes integer, classes_with_tkb integer, entries integer, stem_entries integer, teachers integer, conflicts integer, current_term text)
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'

               where o.type = 'school' and o.id in (select public.my_org_tree())),
       cf as (select x.org_id, count(*)::int as n from public.timetable_conflicts() x group by x.org_id)
  select sc.id, sc.name, sc.parent_name,
    (select count(*) from classes c where c.org_id = sc.id)::int,

    (select y.name || ' · ' || y.term from academic_years y where y.org_id = sc.id and y.is_current limit 1)
  from sc order by sc.name;

-- ##### func training_check_in(p_session uuid)
CREATE OR REPLACE FUNCTION public.training_check_in(p_session uuid)
 RETURNS text
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare v_teacher uuid := public.my_teacher_id(); v_row training_attendance;
begin
  if v_teacher is null then raise exception 'Tài khoản chưa liên kết hồ sơ giáo viên' using errcode = '42501'; end if;
  select * into v_row from training_attendance where session_id = p_session and teacher_id = v_teacher;
  if not found then raise exception 'Bạn không có tên trong danh sách buổi tập huấn này' using errcode = '42501'; end if;
  if not public.training_session_open(p_session) then
    raise exception 'Chỉ điểm danh được trong giờ buổi tập huấn' using errcode = '22023';
  end if;
  if v_row.present is true then return 'already'; end if;
  update training_attendance set present = true where session_id = p_session and teacher_id = v_teacher;
  return 'ok';
end $function$
;
-- ##### func training_session_open(p_session uuid)
CREATE OR REPLACE FUNCTION public.training_session_open(p_session uuid)
 RETURNS boolean
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  select exists (select 1 from training_sessions s where s.id = p_session
    and now() between s.starts_at - interval '15 minutes' and s.ends_at + interval '15 minutes')
$function$
;
-- ##### func user_has_perm(p_user uuid, p_module text, p_action text)
CREATE OR REPLACE FUNCTION public.user_has_perm(p_user uuid, p_module text, p_action text)
 RETURNS boolean
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  select public.user_has_perm(p_user, p_module, p_action, (select org_id from profiles where id = p_user))
$function$
;
-- ##### func user_has_perm(p_user uuid, p_module text, p_action text, p_org uuid)
CREATE OR REPLACE FUNCTION public.user_has_perm(p_user uuid, p_module text, p_action text, p_org uuid)
 RETURNS boolean
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  with mine as (select r.id as role_id, r.key, r.org_id from memberships m join roles r on r.id = m.role_id
                where m.user_id = p_user and m.org_id = p_org)
  select exists (select 1 from mine where org_id is null and key in ('principal','it_admin'))
      or exists (
        select 1 from mine m
        left join org_role_permissions o on o.role_id = m.role_id and o.module = p_module and o.org_id = p_org
        left join role_permissions d on d.role_id = m.role_id and d.module = p_module
        where case p_action
          when 'view' then coalesce(o.can_view, d.can_view, false)
          when 'edit' then coalesce(o.can_edit, d.can_edit, false)
          when 'create' then coalesce(o.can_create, d.can_create, false)
          when 'delete' then coalesce(o.can_delete, d.can_delete, false)
          else false end)
$function$
;
-- ##### func user_roles_write()
CREATE OR REPLACE FUNCTION public.user_roles_write()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO 'public'
AS $function$
begin
  if tg_op = 'INSERT' then
    insert into memberships (user_id, org_id, role_id)
      values (new.user_id, coalesce((select org_id from profiles where id = new.user_id), public.my_org_id()), new.role_id);
    return new;
  else
    delete from memberships m where m.user_id = old.user_id and m.role_id = old.role_id
      and m.org_id = (select org_id from profiles where id = old.user_id);
    return old;
  end if;
end $function$
;
